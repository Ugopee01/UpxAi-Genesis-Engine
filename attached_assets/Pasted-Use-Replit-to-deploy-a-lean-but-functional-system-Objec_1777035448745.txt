app/trading.py

import pandas as pd
import requests

def fetch_data(symbol="BTCUSDT"):
    url = f"https://api.binance.com/api/v3/klines?symbol={symbol}&interval=1h&limit=100"
    data = requests.get(url).json()
    df = pd.DataFrame(data)
    df[4] = df[4].astype(float)  # close price
    return df

def get_signal(symbol):
    df = fetch_data(symbol)
    closes = df[4]

    rsi = (closes.diff().clip(lower=0).rolling(14).mean() /
           closes.diff().abs().rolling(14).mean()) * 100

    if rsi.iloc[-1] < 30:
        return {"signal": "BUY"}
    elif rsi.iloc[-1] > 70:
        return {"signal": "SELL"}
    else:
        return {"signal": "HOLD"}

def execute_trade(symbol):
    signal = get_signal(symbol)
    # simulate execution (no real trade yet)
    return {"executed": signal}
